/** * ระบบตรวจนับสต็อกสินค้า — Google Apps Script API * ------------------------------------------------- * วางไฟล์นี้ใน Apps Script ที่ผูกกับ Google Sheet ของคุณ (Extensions > Apps Script) * แล้ว Deploy เป็น Web App (ดูขั้นตอนใน README.md) * * โครงสร้างคอลัมน์ในชีต (แถวที่ 1 เป็น Header): * A: ลำดับ (Index) * B: ชื่อสินค้า (Product Name) * C: รหัสสินค้า (Product Code / SKU) * D: Serial Number (S/N) * E: สถานะการตรวจนับ (Status) * F: เวลาตรวจนับ (Counted At) — ระบบเติมให้อัตโนมัติ ไม่ต้องกรอกล่วงหน้า */ const SHEET_NAME = "Stock"; // เปลี่ยนชื่อให้ตรงกับชื่อชีตของคุณ const STATUS_COUNTED = "ตรวจนับแล้ว"; const STATUS_PENDING = "ยังไม่ตรวจนับ"; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) throw new Error("ไม่พบชีตชื่อ '" + SHEET_NAME + "'"); return sheet; } function formatTimestamp_(value) { if (!value) return ""; if (Object.prototype.toString.call(value) === "[object Date]") { return Utilities.formatDate(value, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); } return String(value); } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function readAllRows_() { const sheet = getSheet_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; const values = sheet.getRange(2, 1, lastRow - 1, 6).getValues(); return values .map(function (row, i) { return { rowNumber: i + 2, // actual row in the sheet index: row[0], name: row[1], code: row[2], serial: String(row[3] || "").trim(), status: row[4] || STATUS_PENDING, countedAt: formatTimestamp_(row[5]) }; }) .filter(function (r) { return r.serial !== ""; }); } /** GET ?action=list -> returns all items */ function doGet(e) { const action = (e.parameter.action || "list").toLowerCase(); try { if (action === "list") { const items = readAllRows_(); return jsonOut_({ ok: true, items: items }); } return jsonOut_({ ok: false, message: "ไม่รู้จัก action: " + action }); } catch (err) { return jsonOut_({ ok: false, message: String(err) }); } } /** * POST body (JSON, sent as text/plain to avoid CORS preflight): * { "action": "count", "serial": "XXXX" } * { "action": "reset" } */ function doPost(e) { try { const body = JSON.parse(e.postData.contents || "{}"); const action = (body.action || "").toLowerCase(); if (action === "count") { return jsonOut_(markCounted_(String(body.serial || "").trim())); } if (action === "reset") { return jsonOut_(resetAll_()); } return jsonOut_({ ok: false, message: "ไม่รู้จัก action: " + action }); } catch (err) { return jsonOut_({ ok: false, message: String(err) }); } } function markCounted_(serial) { if (!serial) return { ok: false, message: "ไม่ได้ระบุ Serial Number" }; const sheet = getSheet_(); const rows = readAllRows_(); const match = rows.find(function (r) { return r.serial === serial; }); if (!match) return { ok: false, message: "ไม่พบข้อมูลสินค้าในระบบ" }; if (match.status === STATUS_COUNTED) { return { ok: true, alreadyCounted: true, message: "สินค้านี้ถูกตรวจนับไปแล้ว", countedAt: match.countedAt }; } const now = new Date(); sheet.getRange(match.rowNumber, 5).setValue(STATUS_COUNTED); // E: สถานะ sheet.getRange(match.rowNumber, 6).setValue(now); // F: เวลาตรวจนับ return { ok: true, countedAt: formatTimestamp_(now) }; } function resetAll_() { const sheet = getSheet_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return { ok: true }; const statusRange = sheet.getRange(2, 5, lastRow - 1, 1); const statusValues = statusRange.getValues().map(function () { return [STATUS_PENDING]; }); statusRange.setValues(statusValues); const timeRange = sheet.getRange(2, 6, lastRow - 1, 1); const timeValues = timeRange.getValues().map(function () { return [""]; }); timeRange.setValues(timeValues); return { ok: true }; }
